# Segmentation(분할)


## segmentation 종류
1. Semantic: 
	- 클래스는 하나의 덩어리로 취급
	- 모델 : DeepLabV3, U-Net, SegFormer
	- 평가지표 : mIoU, Pixel Accuracy, Dice
2. Instance: 
	- 같은 클래스라도 개별 객체로 분리
	- 모델: YOLOxx-seg
	- 평가지표: Mask mAP
3. Panoptic: 
	- 배경은 시멘틱(도로,하늘), 객체(차,사람)는 인스턴스로 통합분할
	- 모델: Mask2Former, Panoptic FPN
	- 평가지표 : PQ(Panoptic Quality)

## 평가지표

모델이 만든 마스크가 실제 정답과 얼마나 일치하는지 수치로 비교

1. mIOU 
	- mean Intersection over Union
	- 두 바운딩 박스가 얼마나 겹치는지를 나타내는 지표
	- IoU = TP / (TP + FP + FN)    <- 혼동행렬
2. Pixel Accuracy
	- 전체 픽셀 중 올바르게 분류한 픽셀의 비율
	- 단독 지표보다 mIoU 등과 함께 확인
3. Dice
	- Dice Coefficient(계수)
	- 특히 의료 영상처럼 대상 영역이 작은 문제에서 유용
4. Mask mAP
	- 각 객체의 마스크가 정답과 얼마나 겹치는지 평가
5. PQ(Panoptic Quality)
	- PQ = SQ × RQ

#### 1. IoU

두 바운딩 박스가 얼마나 겹치는지를 나타내는 지표

**계산공식**

$IoU = \frac{\text{교집합 면적}}{\text{합집합 면적}}$

- 교집합(Intersection): 두 상자가 겹치는 영역
- 합집합(Union): 두 상자가 차지하는 전체 영역을 합친 것

    print('파일이 존재하지 않아 가상 이미지 생성')
    return np.full((600,800,3), 245, dtype=np.uint8)

#------------------------------------------------
# 두 경게 상자(바운딩 박스)간의 IoU(Intersection over Union)와 교집합 좌표, 면적 계산
#------------------------------------------------
# (x1,y1): 좌측 상단 꼭지점 좌표
# (x2,y2): 우측 상단 꼭지점 좌표
# IoU : 교집합면적 / 합집합 면적
# box_a: Ground Truth(정답상자), box_b: Prediction(예측 상자)
# 리턴값(IoU, inter_box, inter_area, union_area)
#   - IoU : 0.0 ~ 1.0 사이의 IoU점수
#   - inter_box: 교집합 영역의 [x1,y1,x2,y2] 좌표
#   - inter_area : 교집합 면적
#   - union_area : 합집합 면적
def calculate_iou(box_a, box_b):
    # 교집힙(intersection) 영역의 좌상단 및 우하단 좌표 계산
    # 교집합 좌상단
    inter_x1 = max(box_a[0], box_b[0])
    inter_y1 = max(box_a[1], box_b[1])
    # 교집합 우하단
    inter_x2 = min(box_a[2], box_b[2])
    inter_y2 = min(box_a[3], box_b[3])

    # 교집합 영역의 w,h 계산
    inter_w = max(0, inter_x2 - inter_x1)
    inter_h = max(0, inter_y2 - inter_y1)

    # 교집합 영역의 면적 계산
    inter_area = inter_w * inter_h

    # 각 상자의 개별 면적(우하단x - 좌상단x) * (우하단y - 좌상단y)
    area_a = (box_a[2] - box_a[0]) * (box_a[3] - box_a[1])
    area_b = (box_b[2] - box_b[0]) * (box_b[3] - box_b[1])

    # 합집합 면젹 = A 면적 + B면적 - 겹친 면적
    union_area = area_a + area_b - inter_area

    # 교집합 면적 / 합집합 면적
    iou = inter_area / union_area if union_area > 0 else 0

    return iou, [inter_x1, inter_y1, inter_x2, inter_y2], inter_area, union_area

from matplotlib import pyplot as plt

def main():
    img = load_or_create_image(image_path)
    #정답 상자
    gt_box = [200,150,480,430]
    #예측 상자
    pred_box = [260,190,540,470]

    iou, inter_box, inter_area, union_area = calculate_iou(gt_box, pred_box)
    print(f'정답상자: {gt_box}')
    print(f'예측상자: {pred_box}')
    print(f'교집합면적: {inter_area}')
    print(f'합집합면적: {union_area}')
    print(f'최종 IoU: {iou}')

    overlay = img.copy()
    if inter_area > 0:
        # (시작좌표),(종료좌표), 색상, -1:내부 전체 채우기
        cv2.rectangle(overlay, (inter_box[0], inter_box[1]), (inter_box[2], inter_box[3]), (0,200,255), -1)
        # overlay *0,4 + img * 0.6 +  투명도
        img = cv2.addWeighted(overlay, 0.4, img, 0.6, 0, img)
    # 초록색박스

    # 파란색박스
    cv2.rectangle(img, (pred_box[0], pred_box[1]), (pred_box[2], pred_box[3]), (235,99,37), 3)

    cv2.putText(img, f'IoU={iou:.4f}', (200,120), cv2.FONT_HERSHEY_SIMPLEX, 1.0, (15,23,42), 3)
    cv2.putText(img, 'Green: Truth | Blue:Prediction | Yellow:OverLap', (200, 520), cv2.FONT_HERSHEY_SIMPLEX, 1.0, (250,250,250), 3)
    output_path = 'output_iou.jpg'

    print(f'결과 이미지 저장:{output_path}')

"""

import cv2
import numpy as np
import os
import sys

from IPython.paths import get_ipython_module_path
IMAGE_PATH = '/content/drive/MyDrive/Colab Notebooks/dataset/images/person1.jpg'
GT_MASK_PATH = '/content/drive/MyDrive/Colab Notebooks/dataset/images/persion1_mask.jpg'

image_path_bus =  '/content/drive/MyDrive/Colab Notebooks/dataset/images/bus.jpg'

from google.colab.patches import cv2_imshow
from ultralytics import YOLO

THRES_HOLD = 0.5

# img_path를 참고하여 정답 마스크와 예측 마스크 생성
# - 예측 마스크: 모폴로지(팽창/침식) 및 위치 시프트 기법을 사용하여 오차 적용
# - img_path : 원본 이미지 경로
# - gt_path : 저장할 정답 마스크 이미지 경로
# - pred_path : 저장할 예측 마스크 이미지 경로
def generate_masks_from_person1(img_path, gt_path, pred_path ):

    img = cv2.imread(img_path)
    h, w = img.shape[:2]
    gt_mask = np.zeros((h, w), dtype=np.uint8)


    #-------------------------------------------------------
    # 결과 마스크 생성
    # 인스턴스 세그멘테이션(Instance Segmentation) 결과에서 사람(person)의 마스크만
    # 추출한 다음, 원본 영상 크기에 맞추고 여러 사람의 마스크를 하나로 합침
    #-------------------------------------------------------
    try:

        # YOLO 세그멘테이션 모델 가중치 다운로드 및 로드
        model = YOLO('yolov8n-seg.pt')
        results = model(img, verbose=False)[0]

        # 검출된 인스턴스 세그멘테이션 마스크가 존재하는지 체크
        if results.masks is not None:
            # zip()은 두 데이터에서 같은 순서에 있는 항목을 하나씩 묶습니다.
            for mask_data, cls_id in zip(results.masks.data, results.boxes.cls):
                # MS COCO 데이터셋: 0:person
                if int(cls_id) == 0:
                    # 05를 기준으로 마스크 이진화 및 크기 조정 ( 0.5이하면 배경, 이상이면 사람영역으로 처리)
                    # cv2.INTER_NEAREST: 최근방 이웃 보간법을 사용하여 경계 흐림 방지
                    m_np = cv2.resize((mask_data.cpu().numpy() > THRES_HOLD).astype(np.uint8),
                                      (w, h), interpolation=cv2.INTER_NEAREST)
                    # 여러명의 인물이 있을 경우 마스크를 합침(비트 OR 연산)
                    gt_mask = np.maximum(gt_mask, m_np)

    except Exception as e:
        sys.exit(f'YOLO 자동 마스크 추출 에러 발생하여 프로그램 종료! 상세 오류: {e}')

    # 정답 마스크 저장 디렉토리가 없으면 자동 생성
    os.makedirs(os.path.dirname(gt_path) or '_', exist_ok=True)

    # 배경(0), 객체(255) 값으로 이미지 저장
    cv2.imwrite(gt_path, gt_mask * 255)

    #-------------------------------------------------------
    # 예측마스크 생성
    # 정답 마스크(Ground Truth Mask)를 일부러 변형하여 가짜 예측 마스크(Prediction Mask)를 만듬. IoU 계산하기 위해
    #-------------------------------------------------------
    # 3가지 오차를 인위적으로 생성하여 예측 마스크에 적용
    # - 상단부: 팽창(dipation) -> 경계가 밖으로 번져 실제보다 과잉 탐지(FP, Red 오차)
    # - 하단부: 침식(Erosion) -> 경계 안쪽으로 파고들어 미탐지 누락(FN, Blue 오차)
    # - 평행이동(Translation Shift) -> 모델의 좌표 추정 흔들림 적용

    # 모폴로지 연산에 사용할 커널
    kernel_dilate = np.ones((5, 5), np.uint8)
    kernel_erode = np.ones((7, 7), np.uint8)

    #cv2.dilate : 마스크 외곽을 팽창시켜 영역 확장(과잉 분할 발생)
    dilated = cv2.dilate(gt_mask, kernel_dilate, iterations=2)
    # cv2.erode: 마스크 외곽을 깍아내여 영역 축소(미탐지 누락 발생)
    eroded = cv2.erode(dilated, kernel_erode, iterations=1)

    pred_mask = dilated.copy()
    # 위쪽 절반은 팽창된 마스크, 아래쪽 절반은 침식된 마스크 조합
    pred_mask[ h //2:, :] = eroded[ h //2:, :]
    # 우측으로 4픽셀, 아래로 2픽셀 이동  -> 겹치는 영역이 줄어들어 IoU가 낮아질 수 있습니다.
    M = np.float32([[1, 0, 4], [0, 1, 2]])
    pred_mask = cv2.warpAffine(pred_mask, M, (w, h), borderValue=0)

    # 예측 마스크 저장
    pred = pred_mask * 255
    cv2.imwrite(pred_path, pred)
    print(f'정답마스크:{gt_path} \n 예측마스크:{pred_path} 저장!')
    return gt_mask,  pred_mask

#-------------------------------------------------------
# 정답 미스크와 예측 마스크를 픽셀 단위로 비교하여 세그멘테이션 지표 구함
# 객체뿐만 아니라 배경까지 포함한 모든 클래스의 IoU를 각각 구한 뒤 평균을 낸 Multi-class Mean IoU
#-------------------------------------------------------

def compute_segmentation_metrics(gt, pred):
    '''
    정답 미스크와 예측 마스크를 픽셀 단위로 비교하여 세그멘테이션 지표 구함

    parameters
        gt : 정답 마스크
        pred : 예측 마스크

    returns
    # Pixel Accuracy : 전체 픽셀 중 올바르게 분류한 픽셀의 비율
    # foreground_iou : 객체 IoU   background_iou: 배경 IoU
    # miou : 객체와 배경의 평균 Iou,
    # dice_score: 다이스 계수(F1-Score)
    # (tp,tn,fp,fn) : 픽셀 단위 혼동 행력
    '''
    tp=np.sum((gt == 1) & (pred == 1))  # TP: 정답, 예측이 모두 객체(1)인 올바른 객체 픽셀
    tn=np.sum((gt == 0) & (pred == 0))  # TN: 정답. 예측이 모드 배경(0)인 올바른 배경 픽셀
    fp=np.sum((gt == 0) & (pred == 1))  # FP: 정답은 0인테 예측은 1인 과잉탐지 픽셀
    fn=np.sum((gt == 1) & (pred == 0))  # FN: 정답은 1인데 예측은 0인 미탐지 픽셀
    total_pixels = tp + tn + fp + fn

    # Pixel Accuracy: 전체 픽셀 중 맞춘 픽셀(TP + TN)의 비율
    pixel_acc = (tp + tn) / max(total_pixels, 1)

    # 객체 클래스  IoU = 교집합 / 합집합 = tp /(tp + fp + fn)
    foreground_iou = tp / max(tp + fp + fn, 1)
    # 배경 클래스  IoU = 교집합 / 합집합 = tn /(tp + fp + fn)
    background_iou = tn / max(tn + fp +  fn, 1)
    # 평균 IoU
    miou = (foreground_iou  + background_iou) / 2.0

    # 다이스 계수 : 정답 마스크와 예측 마스크가 겹치는 정도를 평가
    dice_score = (2.0 * tp) / max(2*tp + fp + fn, 1)

    return pixel_acc, foreground_iou, background_iou, miou, dice_score, (tp,tn,fp,fn)

import matplotlib.pyplot as plt
import seaborn as sns

def plot_confusion_matrix(tp, tn, fp, fn):
    # 혼동 행렬 array 구성 (2x2)
    # 구조:
    # [ TN, FP ] (실제 배경 0 일 때, 예측 배경 0 / 예측 객체 1)
    # [ FN, TP ] (실제 객체 1 일 때, 예측 배경 0 / 예측 객체 1)
    cm = np.array([
        [tn, fp],
        [fn, tp]
    ])

    # 각 칸에 들어갈 텍스트 레이블 생성
    group_names = ['TN (True Background)', 'FP (Over-segmentation)', 'FN (Missed Object)', 'TP (True Object)']
    group_counts = [f"{value:,}" for value in cm.flatten()]
    group_percentages = [f"{value:.2f}%" for value in (cm.flatten() / np.sum(cm)) * 100]

    labels = [f"{v1}\n{v2}\n{v3}" for v1, v2, v3 in zip(group_names, group_counts, group_percentages)]
    labels = np.asarray(labels).reshape(2, 2)

    # 시각화 설정
    plt.figure(figsize=(8, 6))
    sns.heatmap(cm, annot=labels, fmt='', cmap='Blues', cbar=True,
                xticklabels=['Predict: Background (0)', 'Predict: Object (1)'],
                yticklabels=['Actual: Background (0)', 'Actual: Object (1)'])

    plt.title('Pixel-level Segmentation Confusion Matrix', fontsize=14, pad=15)
    plt.ylabel('Actual Class', fontsize=12)
    plt.xlabel('Predicted Class', fontsize=12)
    plt.tight_layout()
    plt.show()

# 결과 마스크와 예측 마스크로 IoU 계산
# 픽셀 단위로 비교하여 세그멘테이션 지표 구함
# 객체(사람) 영역만 중요하게 평가하는 Single-class IoU
#-------------------------------------------------------
def calculate_iou(gt_path, pred_path):
    # 마스크 이미지 불러오기 (그레이스케일)
    gt_img = cv2.imread(gt_path, cv2.IMREAD_GRAYSCALE)
    pred_img = cv2.imread(pred_path, cv2.IMREAD_GRAYSCALE)

    if gt_img is None or pred_img is None:
        print("마스크 이미지를 불러올 수 없습니다. 경로를 확인해 주세요.")
        return None

    # 0보다 큰 값을 가진 픽셀을 객체(True/1) 영역으로 이진화
    gt_binary = gt_img > 0
    pred_binary = pred_img > 0

    # 교집합(Intersection)과 합집합(Union) 계산
    intersection = np.logical_and(gt_binary, pred_binary).sum()
    union = np.logical_or(gt_binary, pred_binary).sum()

    # 합집합이 0인 경우 처리 (분모가 0이 되는 것을 방지)
    if union == 0:
        return 0.0

    iou = intersection / union
    return iou

# IoU 계산 실행
iou_result = calculate_iou(GT_MASK_PATH, PRED_MASK_PATH)
if iou_result is not None:
    print(f"Calculated IoU: {iou_result:.4f} ({iou_result * 100:.2f}%)")
