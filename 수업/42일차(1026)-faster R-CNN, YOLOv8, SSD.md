```python

import cv2
import numpy as np
import torch
import torchvision
from torchvision.transforms import functional as F
import os

IMAGE_PATH = '/content/drive/MyDrive/data/pedestrians.jpg'

#탐지 임계값
# 모델이 예측한 신뢰도가 0.5이상인 객체만 결과로 채틱
# - 값이 너무 낮으면 잘못된 탐지가 늘어나고, 너무 높으면 객체를 놓칠 수 있음
CONFIDENCE_THRESHOLD = 0.5

# 이미지가 존재하면 사용하고 없으면 더비 이미지 생성
def load_or_create_image(path):
  if os.path.exists(path):
    img = cv2.imread(path)
    if img is not None:
      return img
  print(f'{path} 파일이 없어 더미 임지 생성!')
  return np.null((600, 800,3), 230, dtype=np.unit8)

from google.colab.patches import cv2_imshow

# Faster R-CNN: 2-Statge 객체 검출 모델
# - RPN(Resion Proposal Network): 특징 맵에서 객체가 있을 법한 후보 영역을 제안
# - ROI Pooling & Fast R-CNN : 후보 영역의 특징을 정밀하게 분류 및 바운딩 박스 조정
# - 1-stage 모델(YOLO, SSD)보다 연산속도는 느리지만 정지 이미지에서 소형 객체 및 경계선 탐지가 우수함
def main():
  device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
  print(f'Faster R-CNN (ResNet50-FPN) 추론:{device}')

  #------------------------------------------------------------
  # # Step 1: Initialize model with the best available weights
  #------------------------------------------------------------
  # COCO 데이터셋(마이크로소프트) 학습 가중치
  weights = torchvision.models.detection.FasterRCNN_ResNet50_FPN_Weights.DEFAULT

  # Resnet-50 백본(핵심적인 중심망)과 FPN(Feature Pyramid Noetwork) 구조가 결합된 R-CNN 모델 생성
  model = torchvision.models.detection.fasterrcnn_resnet50_fpn(weights=weights)

  #모델을 연산장치로 전송하고 평가모드로 전환
  model.to(device).eval()

  # COCO 데이터셋의 클래스 이름 목록 추출
  categories = weights.meta['categories']
  print(categories)

  #------------------------------------------------------------
  # Step 2: Initialize the inference transforms
  #------------------------------------------------------------
  bgr_img = load_or_create_image(IMAGE_PATH)
  rgb_img = cv2.cvtColor(bgr_img, cv2.COLOR_BGR2RGB)
  #------------------------------------------------------------
  # Step 3: Apply inference preprocessing transforms
  #------------------------------------------------------------
  # numpy 배열값을 Python 텐서로 변환 -> 배치 차원 추가 ->
  # 입력 데이터를 모델과 동일한 연산 장치로 이동
  input_tensor = F.to_tensor(rgb_img).unsqueeze(0).to(device)


  #------------------------------------------------------------
  # # Step 4: Use the model and print the predicted category
  #------------------------------------------------------------
  # 모델 평가중에 기울기 계산 및 연산 그래프 생성을 생략
  # - 메모리 사용량을 줄이면서 추론 속도 높임
  with torch.no_grad():
    predictions = model(input_tensor)[0] # 딕셔너리 결과값
  # GPU 에 있는 텐서를 CPU 메모리로 복사한 뒤, OpenCV/Numpy 연산
  boxes = predictions['boxes'].cpu().numpy() # 검출괸 객체의 바운딩 박스 좌표(x1, x2, y1, y2)
  labels = predictions['labels'].cpu().numpy() # 검출된 객체의 클래스 정수 ID
  scores = predictions['scores'].cpu().numpy() #검출된 객체에 대한 모델의 신뢰도

  detected_count = 0

  # 바운딩 박스, 클래스 ID, 신뢰도를 zip을 사용하여 한쌍으로 받음
  for box, label_id , score in zip(boxes, labels, scores):
    # 임계값 이상만 객체로 인식
    if score >= CONFIDENCE_THRESHOLD:
      detected_count +=1

    # 박스 좌표를 픽셀 좌표로 변환
    x1,y1,x2,y2 = box.astype(int)
    # 인덱스값에 대응하는 COCO 클래스명 조회
    class_name = categories[label_id]
    # 원본 이미지에 바운딩 빅스 그리기
    cv2.rectangle(bgr_img, (x1,y1) , (x2,y2), (235, 99, 37),2)
    # 텍스트 출력 위치
    text_pos = (x1, max(y1-8, 15))
    # 바운딩 박스 위에 텍스트 출력
    print( f'{class_name}:{score:.2f}')
    cv2.putText(bgr_img, f'{class_name}:{score:.2f}', text_pos,
                cv2.FONT_HERSHEY_SIMPLEX, 0.6, (235, 99, 37), 2 )
  cv2_imshow(bgr_img)

if __name__ == '__main__':
  main()
```

# -*- coding: utf-8 -*-
"""cv1026_2_YOLO8.ipynb

import cv2
import time
import os
from ultralytics import YOLO

VIDEO_PATH = '/content/drive/MyDrive/data/solidWhiteRight.mp4'
MODEL_WEIGHTS = 'yolov8n.pt'

def main():
  model = YOLO(MODEL_WEIGHTS)
  cap = cv2.VideoCapture(VIDEO_PATH)
  widht = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
  hehgit =  int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
  fps = int(cap.get(cv2.CAP_PROP_FPS)) or 30

  output_path = 'output_yolo8n_video.mp4'  # 결과 영상 저장 파일명
  fourcc = cv2.VideoWriter_fourcc(*'mp4v') # 인코딩 코덱
  # 처리된 프레임을 새 비디오 파일로 저장할 파일 오픈
  #colab  또는 jupiter 환경에서는 실시간 윈도우창 보는게 힘듬. 파일로 저장해서 결과 확인
  writer = cv2.VideoWriter(output_path, fourcc, fps, (widht, hehgit))

  start_time_ = time.time()
  print('YOLO 실시간 객체 탐지')
  frame_count = 0
  while cap.isOpened():
    ret, frame = cap.read()
    frame_count += 1
    if not ret:
      break
    start_time = time.time()
    # 객체 탐지 수행
    results = model(frame, verbose = False)[0]
    # 1프레임 처리에 걸린 시간을 계산하여 초당 프레임 수 계산
    infer_fps = 1.0/max(time.time()-start_time, 1e-6)  # 프레임 처리속도가 너무 빨라서 0이 되는 것을 방지하기 위해 max사용
    # 객체(물체)를 탐지한 결과를 시각화하여 그림으로 그림
    # 탐지돈 모든 객체의 바운딩 박스, 클래스명, 신뢰도가 있는 이미지 반환
    annotated_frame = results.plot()
    cv2.putText(annotated_frame, f'FPS:{infer_fps:.1f}',
                (20,45), cv2.FONT_HERSHEY_SIMPLEX, 1.0, (0,355,0),3)

    writer.write(annotated_frame)
  end_time_ = time.time()
  cap.release()
  writer.release()
  print((end_time_ - start_time_)/1000)
  print(f'총{frame_count} 프레임 처리 완료, 결과 동영상 저장:{output_path}')

if __name__ == '__main__':
  main()
# CPU 로 실행하는 경우에 FPS 6
# GPU 로 실행하는 경우에 FPS 340  으로 속도가 많이 차이남

# -*- coding: utf-8 -*-
"""cv1026_3_SSD.ipynb

Automatically generated by Colab.

Original file is located at
    https://colab.research.google.com/drive/1lZiJU0xsUS5UpwybJPCjJLWBetvS9BDw

# SSD(Single Shot MultiBox Detector)
- 2016sus Wei Liu 등이 발표한 1-staghe 객체 탐지 모델
- 이미지를 신경망에 한 번(single shot) 통과시키는 것으로 객체의 위치와 클래스를 동시에 예측

## SSD 장점:
1. 빠른 속도
      - 1-stage 방식이므로 Faster R-CNN 보다 속도에 유리
      - 초반엔 YOLO 보다 더 좋았음
2. 다중 스케일 특징 맵 : 다양한 크기의 객체를 잘 포착함
3. 다양한 크기의 기본 박스를 제공함
      - 다양한 크기와 비율을 가진 디폴트 박스를 이용해 특징맵에서 바운딩 박스를 추출
      - 예측을 위해 네트워크가 각 디폴트박스에서 클래스 점수와 사물 모양에 잘 맞는 박스를 만들어 냄
4. 속도와 정확도의 균형
    - YOLO 계열마늠 빠르면서(현재는 아님) Faster R-CNN 에 근접한 정확도
    - 모바일/임베디드용으로 널리 사용

## SSD 단점
1. 입력을 300 X 300으로 축소하기 때문에 아죽 작은 객체에는 약함

## SSD 모델과 가중치
- ssd300_vgg16,  SSD300_VGG16_Weights
"""

import cv2
import torch
import torchvision
from torchvision.transforms import functional as F
import time
import os

VIDEO_PATH = '/content/drive/MyDrive/data/person_bicycle_car.mp4'
CONFIDENCE_THRESHOLD = 0.5

# cv1026-1_ResNet.ipynb 파일과 유사함
def main():
  if not os.path.exists(VIDEO_PATH):
    print(f'동영상 파일이 존재하지 않습니다.')
    return

  # 동영상 정보(fps, width, height)
  cap = cv2.VideoCapture(VIDEO_PATH)
  widht = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
  hehgit =  int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
  fps = int(cap.get(cv2.CAP_PROP_FPS)) or 30
  fourcc = cv2.VideoWriter_fourcc(*'mp4v') # 인코딩 코덱

  # 결과 동영상 파일 오픈
  output_path = 'output_sdd_video.mp4'  # 결과 영상 저장 파일명
  writer = cv2.VideoWriter(output_path, fourcc, fps, (widht, hehgit))

  device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')

  #----------------------------------------
  # 1단계; 학습된 가중치를 가지는 모델 초기화
  #-----------------------------------------
  # COCO 데이터셋으로 학습된 SSD300(VGG16 백본) 가중치
  weights = torchvision.models.detection.SSD300_VGG16_Weights.DEFAULT

  # 모델생성 -> 연산 장치 이동 -> 평가모드 전환
  model = torchvision.models.detection.ssd300_vgg16(weights=weights).to(device).eval()

  # 클래스 ID를 클래스명으로 매핑
  categories = weights.meta['categories']

  while cap.isOpened():
    ret, frame = cap.read()
    if not ret:
      break

    start_time = time.time()

  #----------------------------------------
  # 2단계; 이미지 전처리 및 텐서 변환
  #----------------------------------------
    # 1. cv2.cvtColor(frame, cv2.COLOR_BGR2RGB): OpenCV의 BGR 이미지를 PyTorch 모델이 사용하는 RGB 순서로 변환합니다.
    # 2. F.to_tensor(...): 이미지를 PyTorch 텐서로 변환하고 픽셀 값을 [0, 1] 범위로 정규화(normalize)합니다.
    # 3. .to(device): 모델 연산을 위해 연산 장치(CPU/GPU)로 텐서를 이동시킵니다.
    # 4. .unsqueeze(0): 배치(batch) 차원을 맨 앞에 추가하여 (채널, 높이, 너비) -> (1, 채널, 높이, 너비) 형태로 만듭니다.
    rgb_tensor = F.to_tensor(cv2.cvtColor(frame,cv2.COLOR_BGR2RGB)).to(device).unsqueeze(0).to(device)

  #--------------------------------------
  # 3단계; 모델을 이용하여 추론하고 예측된 분류결과 출력
  #--------------------------------------
    with torch.no_grad():   # 기울기 계산 X
      predictions = model(rgb_tensor)[0] # 딕셔너리 결과값

    # 초당 프레임 처리 수
    fps_curr = 1.0/max(time.time()-start_time, 1e-6)

    # 박스와 라벨 표시
    for box, label_id , score in zip(predictions['boxes'], predictions['labels'], predictions['scores']):
      # 임계값 이상만 객체로 인식
      if score >= CONFIDENCE_THRESHOLD:
        x1,y1,x2,y2 = box.int().cpu().numpy()
        cv2.rectangle(frame, (x1,y1) , (x2,y2), (235, 99, 37),2)
        cv2.putText(frame, f'{categories[label_id]}:{score:.2f}', (x1, max(y1-8, 15)),
                          cv2.FONT_HERSHEY_SIMPLEX, 0.6, (12, 88, 234), 2 )
    writer.write(frame)
  #--- end of while cap.isOpened(): --

  cap.release()
  writer.release()
  print(f'SSD(single Shot MultiBox Detection) 추론 완료:', output_path)
#--- end of main(): --

if __name__ == '__main__':  
  main()
