#### 전체 소스

```python
# 결측치 정보
missing_df = pd.DataFrame({'결측치 개수': df.isnull().sum(),
                           '결측치 비율': df.isnull().mean() * 100})
missing_df = missing_df[missing_df['결측치 개수'] > 0].sort_values('결측치 개수', ascending=False)
display(missing_df)

# 결측치 시각화
plt.figure(figsize=(12,6))
sns.barplot(x=missing_df.index, y=missing_df['결측치 개수'])
plt.title('Titanic 결측치 개수')
plt.xlabel('컬럼')
plt.ylabel('결측치 개수')
plt.xticks(rotation=45)
plt.tight_layout()
plt.show()

# 통계적 방법으로 Age 결측치 보정
stat_df = df.copy()
age_missing_before = stat_df['age'].isnull().sum()
print(f'보정 전 Age 결측치 : {age_missing_before} ')

# 성별 + 객실 등급별 중앙값. age가 NaN인 곳만 해당 그룹의 중앙값으로 대체
# age가 비어 있으면, 그 승객과 성별 및 객실 등급이 같은 승객들의 나이 중앙값으로 채워라.
stat_df['age'] = stat_df['age'].fillna(stat_df.groupby(['sex', 'pclass'])['age'].transform('median'))

age_missing_after = stat_df['age'].isnull().sum()
print(f'보정 후 Age 결측치 : {age_missing_after} ')

# RandomForest 를 이용한 Age 보정
ml_df = df.copy()

# Sex 숫자 변환
ml_df['sex_encoded'] = ml_df['sex'].map({'male':0, 'femail':1})

# 사용할 feature
age_features = ['pclass', 'sex_encoded', 'sibsp', 'parch', 'fare']

# Age가 존재하는 데이터
age_train = ml_df[ml_df['age'].notnull()].copy()

# Age가 없는 데이터
age_missing = ml_df[ml_df['age'].isnull()].copy()
print(f'Age 학습데이터 : {len(age_train)}')
print(f'Age 예측데이터 : {len(age_missing)}')

rf_age_model = RandomForestRegressor(n_estimators=300,
                               max_depth=12,
                               min_samples_leaf=3,
                               random_state=42,
                               n_jobs=-1)
rf_age_model.fit(age_train[age_features], age_train['age'])

# RandomForest Age 예측
predicted_age = rf_age_model.predict(age_missing[age_features])
print(len(predicted_age))

# Random Forest 결과 적용
rf_df = df.copy()
rf_df.loc[rf_df['age'].isnull(), 'age'] = predicted_age
print(f'Random Forest 보정 후 Age 결측치 : {rf_df["age"].isnull().sum()}')

# 통계적 방법 VS Random Forest 비교
comparison = pd.DataFrame({'통계적 중앙값': stat_df.loc[df['age'].isnull(),'age'].values,
                          'Random forest': rf_df.loc[df['age'].isnull(), 'age'].values})
comparison['차이'] = (comparison['통계적 중앙값'] - comparison['Random forest']).abs()
display(comparison.head(20))

print('[보정결과요약]')
print(f'통계적 방법 평균 Age : ' f'{comparison['통계적 중앙값'].mean():.2f}')
print(f'Random Forest 평균 Age : ' f'{comparison['Random forest'].mean():.2f}')
print(f'두 방법 차이 평균 : ' f'{comparison['차이'].mean():.2f}')

# Age 보정 결과 시각화
plt.figure(figsize=(6,3))  # 단위 Inch
sns.histplot(comparison['통계적 중앙값'], kde=True, label='통계적 중앙값')
sns.histplot(comparison['Random forest'], kde=True, label='Random Forest')
plt.title('Age 결측치 보정 결과 비교')
plt.xlabel('예측 Age')
plt.ylabel('개수')
plt.legend()
plt.tight_layout()
plt.show()
```

### 데이터 전처리 순서

1. 데이터 구조 확인
2. 결측치 정보 확인
    1. 결측치가 있는 컬럼과 결측치 갯수 확인하고 시각
3. 결측치 처리
    1. age 컬럼 결측치 보정
        1. 중간값(mediam)으로 보정
        2. RandomForest 를 이용
        3. 보정결과 비교
        4. 보정 결과 시각화
    2. embaked 컬럼 결측치 제거
        1. 최빈값(mode() 으로 보정
4. 이상치 분석
    1. IQR
5. 이상치 처리
6. capping
