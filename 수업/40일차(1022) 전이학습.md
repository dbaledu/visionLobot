## 전이학습과 처음부터 학습하기 비교

|항목|처음부터 학습|전이학습|
|---|---|---|
|초기 가중치|무작위 초기화|사전 학습된 가중치|
|학습 데이터|일반적으로 많이 필요|비교적 적은 데이터로도 시작 가능|
|학습 시간|상대적으로 길 수 있음|상대적으로 짧을 수 있음|
|성능|데이터와 학습에 따라 달라짐|적절한 사전 학습 모델을 선택하면 유리|
|주요 목적|새로운 모델 지식 학습|기존 지식 재사용|

#### 딥러닝 모델 구조
- 특징 추출부(Feature Extractor) : 이미지에서 선, 모서리, 질감, 형태 등 유용한 특징을 학습
- 분류부(Classification Head) : 추출한 특징을 이용해 자동차, 사람 등 클래스를 판별

#### 전이 학습의 두가지 방법
1. 특성 추출 (Feature Extraction)
     - 기존 모델의 특징 추출부를 고정하고, 새로운 분류부 등 필요한 부분만 학습합니다.
     - 학습이 빠르고 데이터가 적을 때 유리합니다.
	 - 기존 가중치 고정
		 - 기존 특징 추출부: 학습하지 않음
	     - 새로운 출력부:  학습함
2. 미세조정 (Fine-tuning)
	- 새로운 데이터의 특성에 더 잘 적응할 수 있습니다.
	- 사전 학습된 모델을 새로운 데이터로 추가 학습하여 모델의 특징 표현을 조정합니다.
		- 일부 층: 고정할 수 있음
		- 선택한 층: 추가 학습

class PytorchCustomDataset(Dataset):
  def __init__(self,
               root_dir= '/content/cats_and_dogs_filtered/train',
               transform=None):
    self.image_abs_path = root_dir     # 학습/검증 디렉토리 절대경로명(full path)
    self.transform = transform          # 데이터 전처리 객체
    self.label_list = os.listdir(self.image_abs_path)  # 해당경로의 파일or폴더 리스트 (cats, dogs )
    self.label_list.sort()
    self.x_list = []             # 이미지파일의 전제경로
    self.y_list = []             # 각 이미지에 대한 라벨 인덱스

    # 클래스 디렉토리 조회 (클래스별로 폴더가 분리되어 있음)
    for label_index, label_str in enumerate(self.label_list):
      # 클래스 디렉토리 경로
      img_path = os.path.join(self.image_abs_path, label_str)
      # 이미지 파일명 목록
      img_list = os.listdir(img_path)
      for img in img_list:
        # index에 맞게 이미지 데이터와 라벨값 저장
        self.x_list.append(os.path.join(img_path, img))
        self.y_list.append(label_index)

  def __len__(self):
    return len(self.x_list)

  # 인덱스에 해당하는 샘플 1개(이미지, 라벳) 반환
  def __getitem__(self, idx):
    image = Image.open(self.x_list[idx])
    if image.mode != 'RGB':
      image = image.convert('RGB')

    if self.transform:
      image = self.transform(image)

    label = self.y_list[idx]
    return image, label

  #클래스 이름 목록을 파일에 저장
  def __save_label_map__(self, dst_text_path='label_map.txt'):
    label_list = self.label_list
    f = open(dst_text_path, 'w')
    for label in label_list:
      f.write(label+'\n')
    f.close()

  # 클래스 개수 반환
  def __num_classes__(self):
    return len(self.label_list)

import torch
from torchvision import models
from torchvision.models import ResNet18_Weights
import torch.nn as nn
import torch.nn.functional as F

#-----------------------
# 전이학습(Transfer Learning) 기반 모델 클래스 정의
# - 고양이 개 분류에 맞는 분류기 추가
#-----------------------
# n.Module: PyTorch에서 신경망 모델을 만들 때 상속하는 기본 클래스
class MODEL(nn.Module):
  def __init__(self, num_classes):
    super().__init__()
    #ImageNet 데이터셋으로 사전학습된 가중치를 가지는 ResNet 18 CNN 모델
    # ResNet 18 모델 기반 : 2015년 마이크로 소프트가 발표한 CNN 모델
    self.network = models.resnet18(weights=ResNet18_Weights.IMAGENET1K_V1)  #(pretrained=True)

    # 이진 클래스 분류기
    self.classifier = nn.Sequential(
        nn.Dropout(),                   # 과적합 방지(0.5)
        nn.Linear(1000, num_classes),   # 완전연결층(1000 -> num_classes)
        nn.Sigmoid()                    # 출력 활성화 함수
    )

  def forward(self, x):
    x = self.network(x)        # 특징 추출
    return self.classifier(x)  # 분류

import torch.optim as optim
import torchvision.transforms as transforms

train_losses, train_accuracy = [], []
val_losses, val_accuracy = [], []

#-----------------------
#학습 메인 함수
#-----------------------
def trainmain():
  #CPU/PU 사용가능 체크
  USE_CUDA = torch.cuda.is_available()
  DEVICE = torch.device('cuda' if USE_CUDA else 'cpu')
  img_width, img_height = 224, 224
  EPOCHS = 12
  BATCH_SIZE = 32

  # 학습용 전처리: 데이터 증강을 하여 과적합 방지
  transform_train = transforms.Compose([
      transforms.Resize((img_width, img_height)),
      transforms.RandomRotation(degrees=15),
      transforms.ToTensor(),
  ])

  # 검증용 전처리: 데이터 증강을 사용하지 않음
  transform_test = transforms.Compose([
      transforms.Resize((img_width, img_height)),
      transforms.ToTensor(),
  ])

  #-----------------------
  # 데이터셋 로드
  #-----------------------
  # 사용자 정의 데이터셋 사용
  TrainDataset = PytorchCustomDataset
  TestDataset = PytorchCustomDataset

  # 학습용/테스트용  데이터셋
  train_data = TrainDataset(root_dir='/content/cats_and_dogs_filtered/train',
                            transform=transform_train)
  test_data = TestDataset(root_dir='/content/cats_and_dogs_filtered/validation',
                            transform=transform_test)

  # 학습용 데이터를 배치크기로 잘라서 가져옴
  train_loader = torch.utils.data.DataLoader(
      train_data,
      batch_size=BATCH_SIZE,
      shuffle=True        # 매 epoch마다 데이터 순서를 섞어 학습 편향 방지
  )
  # 테스트용 데이터를 배치크기로 잘라서 가져옴
  test_loader = torch.utils.data.DataLoader(
      test_data,
      batch_size=BATCH_SIZE,
      shuffle=False
  )

  #-----------------------
  # 모델 준비
  #-----------------------
  train_data.__save_label_map__()            # 클래스 이름 목록을 label_map.txt에 저장
  num_classes = train_data.__num_classes__() # 클래스 개수
  model = MODEL(num_classes).to(DEVICE)      # 모델 객체 생성 및 장치 설정. 2개의 클래스를 분류하는 모델 생성
  model_str = 'Pytorch_Calssification_Model.pt' # 저장할 모델 파일명
  # 최적화 함수와 학습률 지정
  # 최적화함수 : 사전 학습된 가중치를 최대한 사용하기 위해 낮은 학습률 적용
  optimizer = optim.Adam(model.parameters(), lr=0.00001)
  # step_size = 10 : 10 에포크마다 기존 학습률에 감마값을 곱함
  # 예) 1~10 epoch: 1e-4  11~20: 1e-5  21~30: 1e-6
  scheduler = optim.lr_scheduler.StepLR(optimizer, step_size=10, gamma=0.1)
  acc = 0.0   # 베스트 모델 저장

  #-----------------------
  # 훈련과 검증: epoch 수만큼 반복
  #-----------------------
  for epoch in range(1, EPOCHS+1):
    #-----------------------
    # 학습모드
    #-----------------------
    model.train()    # 학습모드
    tr_loss = 0.0    # 누적손실
    tr_correct = 0.0 # 정답학습 샘플 갯수

    for data, target in train_loader:
      data, target = data.to(DEVICE), target.to(DEVICE)
      # 누적된 기울기 초기화
      optimizer.zero_grad()
      # 순전파
      output = model(data)
      # 역전파에 사용할 loss값
      loss = F.cross_entropy(output, target)
      # 손실값 누적
      tr_loss += F.cross_entropy(output, target, reduction='sum').item()
      # 예측 인덱스값
      pred = output.max(1, keepdim=True)[1]
      # 정답 개수 누적
      tr_correct += pred.eq(target.view_as(pred)).sum().item()
      # 역전파
      loss.backward()
      # 최적화
      optimizer.step()

    # 1 epoch이 끝나면 스케쥴러 갱신
    scheduler.step()
    # 에포크 평균 손실 및 정확도
    tr_ep_loss = tr_loss / len(train_loader.dataset)
    tr_ep_accuracy = 100. * tr_correct / len(train_loader.dataset)

    #-----------------------
    # 평가모드
    #-----------------------
    model.eval()       # 평가 모드
    te_loss = 0.0      # 누적 테스트 손실
    te_correct = 0.0   # 정답 테스트 샘플 갯수

    with torch.no_grad():  # 기울기 계산 비활성화
      for data, target in test_loader:
        data, target = data.to(DEVICE), target.to(DEVICE)

        # 순전파
        output = model(data)
        # 역전파에 사용할 loss값
        loss = F.cross_entropy(output, target)
        # 손실값 누적
        te_loss += F.cross_entropy(output, target, reduction='sum').item()
        # 예측 인덱스값
        pred = output.max(1, keepdim=True)[1]
        # 정답 개수 누적
        te_correct += pred.eq(target.view_as(pred)).sum().item()
        # end of for batch

      # 에포크 평균 손실 및 정확도
      te_ep_loss = te_loss / len(test_loader.dataset)
      te_ep_accuracy = 100. * te_correct / len(test_loader.dataset)
      #------end of with

    # 결과 출력
    print(f'[{epoch}] train Loss:{tr_ep_loss} \
            Train Accuracy:{tr_ep_accuracy}% \
            Test Loss:{te_ep_loss}, Test Accuracy:{te_ep_accuracy}%')

    # 검증 정확도가 이전 최고 기록보다 높아지면 파일로 저장
    if acc < te_ep_accuracy:
      acc = te_ep_accuracy
      torch.save(model.state_dict(), model_str)
      print(f'Model Saved:{model_str}')

    # 시각화를 위해서 리스트에 저장
    train_losses.append(tr_ep_loss)
    train_accuracy.append(tr_ep_accuracy)
    val_losses.append(te_ep_loss)
    val_accuracy.append(te_ep_accuracy)
    #---- end of for epoch


#-----------------------
# 메인함수 호출
#-----------------------
trainmain()

def print_result(inference_result, class_map):
  # 가장 높은 점수를 가진 결과(index)를 클래스 이름으로 변환
  class_text = class_map[np.argmax(inference_result)]

# 이미지를 분류모델로 추론하는 함수
def inference_image(opencv_image, transform_info, model, DEVICE):
  # numpy array -> PIL 이미지 변환
  image = Image.fromarray(opencv_image)
  # 학습할 때와 같은 전처리 적용
  image_tensor = transform_info(image)
  # 맨 앞에 새로운 차원 추가
  image_tensor = image_tensor.unsqueeze(0)
  # 연산에 관련된 모든 데이터는 동일한 H?W 메모리에 존재해야함
  image_tensor = image_tensor.to(DEVICE)
  # 추론 실행
  result = model(image_tensor)
  return result
