### 이미지 증강
 이미지 증강은 훈련 데이터셋의 크기를 인위적으로 확장하는 기술로, 이미지에 다양한 변형(회전, 좌우 반전, 밝기 조정, 노이즈 추가 등)을 적용합니다. 이를 통해 모델은 더욱 다양한 데이터 시나리오에 노출될 수 있습니다. 
 결과적으로 과적합을 방지하고, 새로운 데이터에 대한 일반화 능력을 향상시킬 수 있습니다. [albumentaion getting start 참조]

```python
# -*- coding: utf-8 -*-

"""cv1021_1.ipynb

Original file is located at  https://colab.research.google.com/drive/1bVVEuRmTbibCR7jRnIABcrMKbQiZMwwJ
  
# Torchvision을 이용한 데이터 증강
- 이미지 처리(비전)는 텐서플로보다 torchvision을 더 많이 사용함
- YOLO 5부터는 pytorch위에서 동작함
- 학습데이터가 적을 경우 증강

## transform(변환)
- 데이터를 조작하여 학습에 적합하게 만들기
  
## 이미지 처리
1. 분류(classification)  
2. 탐지(detect)  
3. 분할(segmentation)  
  
비전(vision): 인간의 시각(눈으로 보고 뇌로 이해하는 능력)  
컴퓨터 비전 : 컴퓨터에 '눈'과 '생각하는 뇌'를 주어, 이미지나 동영상 같은 시각적 데이터를 컴퓨터가 보고(See), 처리하고(Process), 이해(Understand)할 수 있게 만드는 기술 분야
"""
  
from PIL import Image
import time
import torch
import torchvision
from torch.utils.data import Dataset
from torchvision import transforms
from matplotlib import pyplot as plt
import cv2
import numpy as np
  
# 이미지 보기
def cv_image_read(image_path):
  print(image_path)
  #데이터를 (h,w,c) 배열로 읽음 이때 채널 순서는 BGR
  return cv2.imread(image_path)
  
#openCV 이미지를 화면에 출력
def show_image(image_path):
  cv_image = cv_image_read(image_path)
  #matplotlib 는 RGB 순서로 처리하므로 변환
  rgb = cv2.cvtColor(cv_image, cv2.COLOR_BGR2RGB)
  plt.figure()
  plt.imshow(rgb)
  plt.show()
  
image_path = '/content/drive/MyDrive/Colab Notebooks/dog_596.jpg'
# cv_image_read 함수를 사용해 이미지 데이터를 numpy array로 읽어옵니다.
show_image(image_path)
  
#-------------------------------------------
# torchvision  데이터셋 클래스 정의(torch.utils.data.dataset을 상속)
#-------------------------------------------
class TorchvisionDataset(Dataset):
  def __init__(self, file_paths, labels, transform=None):
    self.file_paths = file_paths    # 이미지 파일경로 리스트
    self.labels = labels            # 각 이미지에 대한 정답 라벨 리스트
    self.transform = transform      # 전처리 및 데이터 증강
  
  # 오버라이딩
  def __len__(self):
    # 데이터셋의 전체 샘플 개수
    return len(self.file_paths)
  
  # 오버라이딩
  # transform이 매번 적용되어 동일한 이미지가 랜덤하게 변함
  def __getitem__(self, idx):
    label = self.labels[idx]         # 튜플형식(이미지, 라벨)
    file_path = self.file_paths[idx] # 이미지 경로
    image = Image.open(file_path)    # 이미지 읽기 (대문자 Image로 수정)
    if self.transform:
      image = self.transform(image)  # 이미지 변환
    return image, label
  
  
# Torchvision 이미지 변형 인스턴스 생성
torchvision_transform = transforms.Compose([
    transforms.Resize((220,220)),         # 크기 변경 220X220
    transforms.RandomCrop(120),           # 220X220 이미지에서 랜덤의 위치에서 120X120 영역을 잘라냄
    transforms.RandomHorizontalFlip(),    # 50% 확률로 좌우 반전(왼쪽을 본다면 오른쪽을 봄)
    transforms.ToTensor()                 # PIL 이미지 -> FloatTensor 변환 및 정규화  [h,w,c]  --> [c,h,w]로 변환
])
  
  
# TorchvisionDataset 클래스 객체 생성
torchvision_dataset = TorchvisionDataset(
    file_paths = [image_path],                # 이미지 수와 라벨수 일치할 것
    labels = [1],
    transform = torchvision_transform
)


# 이미지 증강 테스트  
for i in range(2):
  sample, _ = torchvision_dataset[0]
  plt.figure()
  # ToPILImage(): FloatTensor -> PIL 이미지로 변환
  # matplotlib 는 FloatTensor 정보를 시각화 처리 못함
  plt.imshow(transforms.ToPILImage()(sample))
  plt.show()
  
  
# 10개의 증강 이미지 생성 및 시각화
imageSize = 10
plt.figure(figsize=(10,6))
for i in range(imageSize):
  sample, _ = torchvision_dataset[0]
  # 2행 5열 서브 플롯
  plt.subplot(2, 5, i+1)
  
  # FloatTensor -> PIL 이미지 변환
  plt.imshow(transforms.ToPILImage()(sample))
  plt.title(f'augmente img {i+1}')
  plt.axis('off')
  
plt.tight_layout()
plt.show()
```
